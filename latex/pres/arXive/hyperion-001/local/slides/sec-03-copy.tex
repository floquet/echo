\subsection{Structure of the \bsvd}

\begin{frame}
\frametitle{\bsvd: Big Picture}
	{\large{
		%
Given a nonzero matrix $\aicmnr$, a matrix with complex entries with $m$ rows, $n$ columns, and matrix rank $0<\rho\le\min\paren{m,n}$, then there exists a decomposition of the form
%
\begin{equation}
	\svdk{*}
\label{eq:svdk}
\end{equation}
		%
	}}
\end{frame}

\begin{frame}
\frametitle{\bsvd: Big Picture}
	{\large{
		%
\begin{enumerate}
  \item column vectors of unitary matrix $\V{}\icnn$ represent an orthonormal span of the row space (\bl{solution space}),
  \item column vectors of unitary matrix $\U{}\icmm$ represent an orthonormal span of the column space (\bl{measurement space}),
  \item entries in the diagonal matrix $\ess{}\in\real{\byy{\rho}}$ contain the singular values; the square root of ordered, nonzero \bl{eigenvalues of the product matrix} $\wx{*}$,
  \item the sabot matrix $\sig{}\irmn$ pads, $\ess{}$, the matrix of singular values with zeros so that is \bl{conformable} with both $\U{}$ and $\V{}$.
\end{enumerate}
		%
	}}
\end{frame}

\begin{frame}
\frametitle{Conformability}
	{\large{
	$$\A{}\in\cmplx{4\times2}_{1}$$
		%
\begin{equation}
	\begin{array}{cccccc}
			%
		&\A{} &= &\U{} &\sig{} &\V{*}\\[5pt]
			%
		&\mat{c|c}{&\\\hline&\\\hline&\\\hline&} &= 
			&\mat{c|c|c|c}{\ccb&\ccr&\ccr&\ccr\\\hline
						   \ccb&\ccr&\ccr&\ccr\\\hline
						   \ccb&\ccr&\ccr&\ccr\\\hline
						   \ccb&\ccr&\ccr&\ccr}
			&\mat{c|c}{\cellcolor{black!20}&0 \\\hline
					0 & 0 \\\hline
					0 & 0 \\\hline
					0 & 0 }
			&\mat{c|c}{\ccb&\ccb\\\hline\ccr&\ccr} \\[35pt]
			%
			& m\times n&& m\times m& \dim\paren{\A{}} & n\times n
			%
	\end{array}
\label{eq:}
\end{equation}
		%
	}}
\end{frame}

	{\large{
		%
\begin{equation}
\begin{split}
	\normts{ \axmb }

		&= \normts{\ess{} \V{*}b - \U{*}b }
\end{split}
\end{equation}
		%
	Use Pythagorus to separate \bl{range} and \rd{null} space components.
		%
	}}

	}}
