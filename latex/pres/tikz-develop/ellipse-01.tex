\documentclass[tikz,border=4pt]{standalone}

\usepackage{tikz}

\begin{document}

\begin{tikzpicture}[
    x=3.2cm,
    y=3.2cm,
    tick/.style={black, line width=0.45pt},
    frame/.style={black, line width=0.7pt},
    agreement/.style={gray!65, line width=0.45pt},
    label/.style={font=\large}
]

% -------------------------------------------------------------------------
% Limits
% -------------------------------------------------------------------------
\def\xmin{-1.05}
\def\xmax{ 1.05}
\def\ymin{-1.05}
\def\ymax{ 1.05}

\def\ticklen{0.02}

% -------------------------------------------------------------------------
% Agreement line
% -------------------------------------------------------------------------
\draw[agreement, gray]
    (-1,-1) -- (1,1);

% -------------------------------------------------------------------------
% Frame
% -------------------------------------------------------------------------
\draw[frame]
    (\xmin,\ymin) rectangle (\xmax,\ymax);

% -------------------------------------------------------------------------
% Tick marks: x-axis positions
% -------------------------------------------------------------------------
\foreach \x in {-1,0,1}{
    \draw[tick]
        (\x,\ymin)
        -- ++(0, 2 * \ticklen);
}

% -------------------------------------------------------------------------
% Tick marks: y-axis positions
% -------------------------------------------------------------------------
\foreach \y in {-1,0,1}{
    \draw[tick]
        (\xmin,\y)
        -- ++( 2 * \ticklen,0);
}

% -------------------------------------------------------------------------
% Tick labels
% -------------------------------------------------------------------------
\node[label,below] at (-1,\ymin) {$-1$};
\node[label,below] at ( 0,\ymin) {$0$};
\node[label,below] at ( 1,\ymin) {$1$};

\node[label,left] at (\xmin,-1) {$-1$};
\node[label,left] at (\xmin, 0) {$0$};
\node[label,left] at (\xmin, 1) {$1$};

% -------------------------------------------------------------------------
% Axis labels
% -------------------------------------------------------------------------
\node[label, below = 10pt ] at (0,\ymin) {$c_0$};
\node[label, left = 10pt ] at (\xmin,0) {$c_\star$};

\end{tikzpicture}

\end{document}
